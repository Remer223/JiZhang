CREATE TABLE t_family (
      id INT PRIMARY KEY AUTO_INCREMENT,
      name VARCHAR(100) NOT NULL COMMENT '家庭名称',
      creator_id INT NOT NULL COMMENT '创建者ID',
      created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (creator_id) REFERENCES t_user(id)
  ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;



CREATE TABLE t_family_member (
      id INT PRIMARY KEY AUTO_INCREMENT,
      family_id INT NOT NULL COMMENT '家庭ID',
      user_id INT NOT NULL COMMENT '用户ID',
      role VARCHAR(50) NOT NULL COMMENT '角色',
      approved TINYINT(1) NOT NULL DEFAULT 0 COMMENT '0=待审批 1=已批准',
      joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (family_id) REFERENCES t_family(id),
      FOREIGN KEY (user_id) REFERENCES t_user(id),
      UNIQUE KEY uk_family_user (family_id, user_id)
  ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
